# Scope: 2-layer, 1 oz outer copper, 1.6 mm FR4, green LPI mask (V196 assumptions; copper
# weight and colour are order options, not fixed by the PCB file).
# Each rule cites the capability row it comes from. Where the project keeps a margin above
# the published minimum, the comment says "project margin" (proposed, not sourced).
(version 1)

# Traces row "Min. track width and spacing (1 oz)": 1- and 2-layer 0.10/0.10 mm.
(rule "JLC_track_width_min"
  (constraint track_width (min 0.10mm)))

# Project margin: keep the general copper clearance at 0.20 mm (twice the 0.10 mm
# published track/space minimum). Not a JLC requirement.
(rule "PROJECT_general_clearance"
  (constraint clearance (min 0.20mm)))

# Traces row "SMD pad to pad clearance (different nets)": 0.15 mm (published capability,
# zero margin). Also Soldermask row "Soldermask bridge", 1 oz green: min pad spacing 0.10 mm.
(rule "JLC_smd_pad_to_pad_different_nets"
  (condition "A.Type == 'Pad' && B.Type == 'Pad' && A.Pad_Type == 'SMD' && B.Pad_Type == 'SMD'")
  (constraint clearance (min 0.15mm)))

# Traces row "Pad to track clearance (1oz)": min 0.10 mm. Kept at the project 0.20 mm
# general clearance above; no separate relaxation proposed.

# Drill rows "Min. Via hole size/diameter": 2-layer 0.15 mm hole / 0.25 mm diameter;
# "Via diameter should be 0.1mm (0.15mm preferred) larger than via hole size".
(rule "JLC_via_hole_min"
  (condition "A.Type == 'Via'")
  (constraint hole_size (min 0.15mm)))
(rule "JLC_via_diameter_min"
  (condition "A.Type == 'Via'")
  (constraint via_diameter (min 0.25mm)))
(rule "JLC_via_annular_min"
  (condition "A.Type == 'Via'")
  (constraint annular_width (min 0.05mm)))

# Traces row "PTH annular ring": 2-layer 1 oz recommended 0.25 mm, absolute minimum 0.18 mm.
(rule "JLC_pth_pad_annular_min"
  (condition "A.Type == 'Pad' && A.isPlated()")
  (constraint annular_width (min 0.18mm)))

# Drill row "Min. Non-plated holes": 0.50 mm.
(rule "JLC_npth_min"
  (condition "A.Type == 'Pad' && !A.isPlated()")
  (constraint hole_size (min 0.50mm)))

# Drill rows "Via Hole-to-Hole Spacing" 0.2 mm and "Pad Hole-to-Hole Spacing" 0.45 mm.
(rule "JLC_hole_to_hole_via"
  (condition "A.Type == 'Via' && B.Type == 'Via'")
  (constraint hole_to_hole (min 0.20mm)))
(rule "JLC_hole_to_hole_pad"
  (condition "A.Type == 'Pad' && B.Type == 'Pad'")
  (constraint hole_to_hole (min 0.45mm)))

# Traces rows "Via hole to Track" 0.2 mm and "NPTH to Track" 0.2 mm.
(rule "JLC_hole_clearance_base"
  (constraint hole_clearance (min 0.20mm)))

# Traces row "PTH to Track": minimum 0.28 mm, 0.35 mm recommended. Applies to copper near
# plated component holes (not vias).
(rule "JLC_pth_hole_clearance"
  (condition "B.Type == 'Pad' && B.isPlated() && B.Pad_Type == 'Through-hole'")
  (constraint hole_clearance (min 0.28mm)))

# Board edge rows: "Copper clearance from routed board edges: >= 0.2 mm".
(rule "JLC_edge_clearance"
  (constraint edge_clearance (min 0.20mm)))

# Drill row "Hole Position Tolerance": +/-0.075 mm. Not expressible as a DRC rule; any
# copper-to-hole gap with nominal margin below 0.075 mm above its minimum is flagged in
# rules/README.md rather than silently accepted.
